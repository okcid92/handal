export async function validateThemeVotingV2(
  themeId: bigint,
  voterId: bigint,
  decision: 'approved' | 'rejected',
  comment: string
): Promise<Theme> {
  
  // 1. Récupérer le thème
  const theme = await prisma.theme.findUnique({
    where: { id: themeId },
    include: { student: true }
  });

  if (!theme) {
    throw new ApiError(404, 'Theme not found');
  }

  if (theme.status !== 'PENDING_VALIDATION') {
    throw new ApiError(409, `Cannot vote on theme with status ${theme.status}`);
  }

  // 2. Déterminer le rôle du votant
  const voter = await prisma.user.findUnique({ where: { id: voterId } });
  if (!voter || !['TEACHER', 'DA', 'ADMIN'].includes(voter.role)) {
    throw new ApiError(403, 'Only TEACHER or DA can vote');
  }

  // 3. Enregistrer le vote
  const updatedTheme = await prisma.theme.update({
    where: { id: themeId },
    data: voter.role === 'TEACHER' ? {
      teacherVote: decision,
      teacherComment: comment,
      teacherVotedAt: new Date()
    } : {
      daVote: decision,
      daComment: comment,
      daVotedAt: new Date()
    }
  });

  // 4. Calculer la décision finale si les 2 ont voté
  if (updatedTheme.teacherVote && updatedTheme.daVote) {
    const finalStatus = 
      updatedTheme.teacherVote === 'approved' && updatedTheme.daVote === 'approved'
        ? 'VALIDATED'
        : 'REJECTED';

    const finalTheme = await prisma.theme.update({
      where: { id: themeId },
      data: { status: finalStatus }
    });

    // Notifier l'étudiant
    await notifyStudent(theme.studentId, {
      type: finalStatus === 'VALIDATED' ? 'THEME_VALIDATED' : 'THEME_REJECTED',
      themeId,
      message: finalStatus === 'VALIDATED'
        ? `Votre thème "${theme.title}" a été validé. Vous pouvez déposer votre mémoire.`
        : `Votre thème "${theme.title}" a été rejeté.`
    });

    return finalTheme;
  }

  return updatedTheme;
}

export async function createDocument(
  studentId: bigint,
  file: File
): Promise<Document> {
  
  // 1. Récupérer le thème VALIDATED de l'étudiant
  const theme = await prisma.theme.findFirst({
    where: {
      studentId,
      status: 'VALIDATED'  // ← SEUL STATUT ACCEPTÉ en v2
    }
  });

  if (!theme) {
    throw new ApiError(403, 
      'You must have a VALIDATED theme before uploading a document. ' +
      'Status must be VALIDATED (both Teacher and DA approved).');
  }

  // 2. Valider le fichier (MIME, taille, etc.)
  const checksum = await calculateSHA256(file);
  
  // 3. Créer le document
  const document = await prisma.document.create({
    data: {
      themeId: theme.id,
      studentId,
      originalName: file.name,
      mimeType: file.type,
      fileSize: BigInt(file.size),
      checksum,
      documentStatus: 'SUBMITTED'
    }
  });

  // 4. Stocker le fichier
  await storeFile(document.id, file);

  // 5. Lancer l'analyse (asynchrone, NON bloquante)
  await queuePlagiarismAnalysis(document.id);

  return document;
}

export async function analyzeDocument(
  documentId: bigint
): Promise<SimilarityReport> {
  
  // 1. Récupérer le document + thème
  const document = await prisma.document.findUnique({
    where: { id: documentId },
    include: { theme: true }
  });

  if (!document) throw new ApiError(404, 'Document not found');

  // 2. Marquer comme ANALYSIS_IN_PROGRESS
  await prisma.document.update({
    where: { id: documentId },
    data: { documentStatus: 'ANALYSIS_IN_PROGRESS' }
  });

  try {
    // 3. Extraire le texte du PDF
    const text = await extractTextFromPDF(document.storagePath);

    // 4. Analyser le plagiat (algos)
    const scores = await analyzePlagiarism(text, {
      referenceDocuments: await getReferenceDocuments(),
      detectedTheme: document.theme.title
    });

    // 5. Créer le rapport
    const report = await prisma.similarityReport.create({
      data: {
        documentId,
        globalSimilarity: new Decimal(scores.combined),
        cosineScore: new Decimal(scores.cosine),
        jaccardScore: new Decimal(scores.jaccard),
        ngramScore: new Decimal(scores.ngram),
        winnowingScore: new Decimal(scores.winnowing),
        lcsScore: new Decimal(scores.lcs),
        styleScore: new Decimal(scores.style),
        simhashScore: new Decimal(scores.simhash),
        semanticScore: scores.semantic ? new Decimal(scores.semantic) : null,
        riskLevel: scores.combined >= 0.70 ? 'high' : scores.combined >= 0.30 ? 'medium' : 'low',
        flagged: scores.combined >= 0.20,
        matchedSources: JSON.stringify(scores.matchedSources || []),
        highlightedSegments: JSON.stringify(scores.highlightedSegments || {}),
        analyzedAt: new Date()
      }
    });

    // 6. Mettre à jour le statut du document
    const finalStatus = scores.combined >= 0.20 ? 'FLAGGED_PLAGIARISM' : 'CLEAN';
    await prisma.document.update({
      where: { id: documentId },
      data: { documentStatus: finalStatus }
    });

    // 7. Notifier Teacher + DA + étudiant
    await notifyMultiple([
      { userId: /* Teacher */, type: 'ANALYSIS_COMPLETE', documentId },
      { userId: /* DA */, type: 'ANALYSIS_COMPLETE', documentId },
      { 
        studentId: document.studentId,
        type: finalStatus === 'FLAGGED_PLAGIARISM' ? 'DOCUMENT_FLAGGED' : 'DOCUMENT_CLEAN',
        documentId,
        score: scores.combined
      }
    ]);

    return report;
  } catch (error) {
    await prisma.document.update({
      where: { id: documentId },
      data: { documentStatus: 'FAILED' }
    });
    throw error;
  }
}

export async function POST(request: Request, { params }: { params: { id: string } }) {
  
  const { decision, comment, mention } = await request.json();
  const user = await requireAuth(request);
  
  if (!['TEACHER', 'DA', 'ADMIN'].includes(user.role)) {
    return errorResponse(403, 'Only TEACHER or DA can appreciate');
  }

  const documentId = BigInt(params.id);

  // 1. Récupérer le document et son rapport
  const document = await prisma.document.findUnique({
    where: { id: documentId },
    include: { similarityReport: true, finalAppreciation: true }
  });

  if (!document) {
    return errorResponse(404, 'Document not found');
  }

  // 2. Vérifier le prérequis (document doit être CLEAN ou FLAGGED analysé)
  if (!['CLEAN', 'FLAGGED_PLAGIARISM'].includes(document.documentStatus)) {
    return errorResponse(409, `Cannot appreciate document with status ${document.documentStatus}`);
  }

  // 3. Upsert FinalAppreciation (créer si n'existe pas, mettre à jour sinon)
  let appreciation = document.finalAppreciation;
  if (!appreciation) {
    appreciation = await prisma.finalAppreciation.create({
      data: { documentId }
    });
  }

  // 4. Enregistrer le vote
  const isTeacher = user.role === 'TEACHER' || (user.role === 'ADMIN' && !appreciation.teacherDecision);
  const isDA = user.role === 'DA' || (user.role === 'ADMIN' && !appreciation.daDecision);

  const updateData: any = {};
  
  if (isTeacher || user.role === 'ADMIN') {
    updateData.teacherId = user.id;
    updateData.teacherDecision = decision;
    updateData.teacherComment = comment;
    updateData.teacherDecidedAt = new Date();
  }
  
  if (isDA || user.role === 'ADMIN') {
    updateData.daId = user.id;
    updateData.daDecision = decision;
    updateData.daComment = comment;
    updateData.daDecidedAt = new Date();
  }

  const updatedAppreciation = await prisma.finalAppreciation.update({
    where: { id: appreciation.id },
    data: updateData
  });

  // 5. Calculer la décision finale si les 2 ont voté
  if (updatedAppreciation.teacherDecision && updatedAppreciation.daDecision) {
    // Règle : les 2 doivent approuver pour approuver
    const canApprove = 
      updatedAppreciation.teacherDecision === 'APPROVED' &&
      updatedAppreciation.daDecision === 'APPROVED';

    const finalDecision = canApprove ? 'APPROVED' : 
      (updatedAppreciation.teacherDecision === 'REJECTED' || 
       updatedAppreciation.daDecision === 'REJECTED' ? 'REJECTED' : decision);

    const finalAppreciation = await prisma.finalAppreciation.update({
      where: { id: appreciation.id },
      data: {
        finalDecision,
        mention: mention || null,
        finalizedAt: new Date()
      }
    });

    // Mettre à jour le statut du document
    const newDocumentStatus = canApprove ? 'APPROVED' : 'REJECTED';
    await prisma.document.update({
      where: { id: documentId },
      data: { documentStatus: newDocumentStatus }
    });

    // Notifier l'étudiant
    const doc = await prisma.document.findUnique({
      where: { id: documentId },
      include: { student: true }
    });
    
    await notifyStudent(doc!.studentId, {
      type: newDocumentStatus === 'APPROVED' ? 'APPRECIATION_APPROVED' : 'APPRECIATION_REJECTED',
      documentId,
      message: newDocumentStatus === 'APPROVED'
        ? `Votre mémoire a été approuvé. Vous êtes autorisé à faire la soutenance.`
        : `Votre mémoire a été rejeté. Veuillez contacter le jury pour les détails.`,
      mention: mention || null
    });

    return successResponse(finalAppreciation);
  }

  return successResponse(updatedAppreciation);
}
